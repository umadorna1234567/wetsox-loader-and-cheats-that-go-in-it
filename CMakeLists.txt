cmake_minimum_required(VERSION 3.21)
project(Wetsox VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Quick QuickControls2)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(Nexus WIN32 src/main.cpp)
if(WIN32)
    set_source_files_properties(assets/wetsox.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/wetsox.ico")
    target_sources(Nexus PRIVATE assets/wetsox.rc)
    target_include_directories(Nexus PRIVATE assets)
endif()
set_target_properties(Nexus PROPERTIES OUTPUT_NAME "Wetsox")
set(qml_files
    qml/Main.qml qml/Theme.qml qml/ActionButton.qml qml/Toggle.qml
    qml/Input.qml qml/Choice.qml qml/SettingsPanel.qml qml/SliderSetting.qml
    qml/ColorSetting.qml qml/GameCard.qml qml/GameMenu.qml qml/FeatureRow.qml
    qml/GameEditor.qml qml/Artwork.qml qml/Glyph.qml qml/NavItem.qml
    qml/WindowBar.qml qml/WindowEdges.qml qml/CoverArt.qml
    qml/ControlPanel.qml qml/CompactControl.qml qml/KeybindPicker.qml qml/ConfigsPanel.qml)
foreach(file IN LISTS qml_files)
    get_filename_component(filename ${file} NAME)
    set_source_files_properties(${file} PROPERTIES QT_RESOURCE_ALIAS ${filename})
endforeach()
qt_add_qml_module(Nexus URI Nexus VERSION 1.0
    QML_FILES ${qml_files}
    SOURCES src/AppState.h src/AppState.cpp src/UiHelpers.h src/UiHelpers.cpp src/GameSession.h src/GameSession.cpp src/ConfigStore.h src/ConfigStore.cpp)
qt_add_resources(Nexus brand_resources PREFIX /brand BASE assets FILES assets/wetsox.ico)
qt_add_resources(Nexus smoke_resources PREFIX /testing FILES tests/Smoke.qml)
target_link_libraries(Nexus PRIVATE Qt6::Quick Qt6::QuickControls2)
target_include_directories(Nexus PRIVATE src)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(theme_tests tests/theme_tests.cpp src/AppState.cpp src/AppState.h)
    target_include_directories(theme_tests PRIVATE src)
    target_link_libraries(theme_tests PRIVATE Qt6::Quick Qt6::Test)
    add_test(NAME theme_tests COMMAND theme_tests)
    set_tests_properties(theme_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    qt_add_executable(interaction_tests tests/interaction_tests.cpp src/UiHelpers.cpp src/UiHelpers.h)
    target_include_directories(interaction_tests PRIVATE src)
    target_link_libraries(interaction_tests PRIVATE Qt6::Quick Qt6::Test)
    if(WIN32)
        target_link_libraries(interaction_tests PRIVATE gdi32 user32)
    endif()
    add_test(NAME interaction_tests COMMAND interaction_tests)
    set_tests_properties(interaction_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_test(NAME ui_smoke COMMAND Nexus --smoke-test)
    set_tests_properties(ui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_QPA_FONTDIR=C:/Windows/Fonts;QT_FORCE_STDERR_LOGGING=1" TIMEOUT 25)
endif()
install(TARGETS Nexus RUNTIME DESTINATION bin)
if(WIN32)
    get_target_property(qmake_path Qt6::qmake IMPORTED_LOCATION)
    get_filename_component(qt_bin "${qmake_path}" DIRECTORY)
    install(CODE "execute_process(COMMAND \"${qt_bin}/windeployqt.exe\" --no-translations --qmldir \"${CMAKE_CURRENT_SOURCE_DIR}/qml\" \"\${CMAKE_INSTALL_PREFIX}/bin/Wetsox.exe\" COMMAND_ERROR_IS_FATAL ANY)")
else()
    qt_generate_deploy_qml_app_script(TARGET Nexus OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
    install(SCRIPT ${deploy_script})
endif()

add_subdirectory(games/farcry5)
add_subdirectory(games/farcry4)
target_include_directories(Nexus PRIVATE games/farcry5/include)
target_compile_definitions(Nexus PRIVATE NOMINMAX)

if(BUILD_TESTING)
 qt_add_executable(session_tests tests/session_tests.cpp src/GameSession.cpp src/GameSession.h)
 target_include_directories(session_tests PRIVATE src games/farcry5/include)
 target_compile_definitions(session_tests PRIVATE NOMINMAX NEXUS_SESSION_TEST)
 target_link_libraries(session_tests PRIVATE Qt6::Quick Qt6::Test)
 add_test(NAME game_session COMMAND session_tests)
 set_tests_properties(game_session PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 50 RUN_SERIAL TRUE)
endif()

if(BUILD_TESTING)
 qt_add_executable(package_config_tests tests/package_config_tests.cpp src/ConfigStore.cpp src/ConfigStore.h src/AppState.cpp src/AppState.h)
 target_include_directories(package_config_tests PRIVATE src)
 target_link_libraries(package_config_tests PRIVATE Qt6::Quick Qt6::Test)
 add_test(NAME packages_and_configs COMMAND package_config_tests)
 set_tests_properties(packages_and_configs PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
