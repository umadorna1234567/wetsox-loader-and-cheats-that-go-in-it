 set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
 add_compile_options("$<$<COMPILE_LANGUAGE:CXX>:/EHsc>")

add_library(nexus_fc5_core STATIC src/targeting.cpp)
target_include_directories(nexus_fc5_core PUBLIC include)
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
add_library(nexus_imgui STATIC third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp third_party/imgui/backends/imgui_impl_win32.cpp third_party/imgui/backends/imgui_impl_dx11.cpp)
target_include_directories(nexus_imgui PUBLIC third_party/imgui third_party/imgui/backends)
target_link_libraries(nexus_imgui PUBLIC d3d11 dxgi d3dcompiler dwmapi)
add_library(NexusFC5 SHARED src/dll_main.cpp src/overlay.cpp src/runtime.cpp)
target_compile_definitions(NexusFC5 PRIVATE FC5_DLL_BUILD UNICODE _UNICODE NOMINMAX)
target_link_libraries(NexusFC5 PRIVATE nexus_fc5_core nexus_imgui minhook bcrypt)
add_executable(NexusGameLoader src/loader.cpp)
target_link_libraries(NexusGameLoader PRIVATE bcrypt)
target_include_directories(NexusGameLoader PRIVATE include)
target_compile_definitions(NexusGameLoader PRIVATE UNICODE _UNICODE NOMINMAX FC5_DLL_FILENAME=L"cheats/farcry5/WetsoxFC5.dll")
install(TARGETS NexusFC5 RUNTIME DESTINATION bin/cheats/farcry5)
set_target_properties(NexusGameLoader PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/backend")
install(TARGETS NexusGameLoader RUNTIME DESTINATION bin/backend)
install(FILES third_party/imgui/LICENSE.txt DESTINATION bin/backend/licenses RENAME imgui.txt)
install(FILES third_party/minhook/LICENSE.txt DESTINATION bin/backend/licenses RENAME minhook.txt)
if(BUILD_TESTING)
 add_executable(fc5_core_tests ../../tests/fc5/core_tests.cpp)
 target_link_libraries(fc5_core_tests PRIVATE nexus_fc5_core)
 add_test(NAME fc5_targeting COMMAND fc5_core_tests)
 add_executable(NexusFC5TestHost ../../tests/fc5/overlay_host.cpp)
 target_include_directories(NexusFC5TestHost PRIVATE include)
 target_compile_definitions(NexusFC5TestHost PRIVATE UNICODE _UNICODE NOMINMAX)

 set_tests_properties(fc5_loading PROPERTIES TIMEOUT 90)
endif()

get_property(nexus_multi_config GLOBAL PROPERTY GENERATOR_IS_MULTI_CONFIG)
if(nexus_multi_config)
 foreach(configuration IN LISTS CMAKE_CONFIGURATION_TYPES)
  string(TOUPPER "${configuration}" configuration_upper)
  set_target_properties(NexusFC5 PROPERTIES "RUNTIME_OUTPUT_DIRECTORY_${configuration_upper}" "${CMAKE_BINARY_DIR}/${configuration}/cheats/farcry5")
 endforeach()
endif()
if(BUILD_TESTING)
 set_target_properties(NexusFC5TestHost PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endif()

set_target_properties(NexusFC5 PROPERTIES OUTPUT_NAME "WetsoxFC5")

set_target_properties(NexusGameLoader PROPERTIES OUTPUT_NAME "WetsoxGameLoader")
foreach(package_file game.json cover.jpg)
 configure_file("package/${package_file}" "${CMAKE_BINARY_DIR}/cheats/farcry5/${package_file}" COPYONLY)
 if(nexus_multi_config)
  foreach(configuration IN LISTS CMAKE_CONFIGURATION_TYPES)
   configure_file("package/${package_file}" "${CMAKE_BINARY_DIR}/${configuration}/cheats/farcry5/${package_file}" COPYONLY)
  endforeach()
 endif()
endforeach()
