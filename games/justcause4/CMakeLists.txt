set(REF "${CMAKE_SOURCE_DIR}/just cause 4 mod/reference/jc4-console-thingy")
file(GLOB FONT_SOURCES "${REF}/deps/fw1/*.cpp")
add_library(wetsox_fw1 STATIC ${FONT_SOURCES})
target_include_directories(wetsox_fw1 PUBLIC "${REF}/deps/fw1")
target_link_libraries(wetsox_fw1 PUBLIC d3d11 d3dcompiler dwrite)
set_target_properties(wetsox_fw1 PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(WetsoxJC4 SHARED src/bridge.cpp src/game.cpp src/render.cpp)
target_include_directories(WetsoxJC4 PRIVATE src "${REF}/src" ../farcry5/include)
target_compile_definitions(WetsoxJC4 PRIVATE WETSOX_JC4 WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
target_compile_options(WetsoxJC4 PRIVATE /EHa)
target_link_libraries(WetsoxJC4 PRIVATE minhook wetsox_fw1 user32)
set_target_properties(WetsoxJC4 PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/cheats/justcause4")
configure_file(package/game.json "${CMAKE_BINARY_DIR}/cheats/justcause4/game.json" COPYONLY)
configure_file(package/cover.jpg "${CMAKE_BINARY_DIR}/cheats/justcause4/cover.jpg" COPYONLY)
install(TARGETS WetsoxJC4 RUNTIME DESTINATION bin/cheats/justcause4)
install(FILES package/game.json package/cover.jpg DESTINATION bin/cheats/justcause4)
install(DIRECTORY package/licenses DESTINATION bin/cheats/justcause4)

if(BUILD_TESTING)
 add_executable(jc4_math_tests ../../tests/jc4_math_tests.cpp)
 target_include_directories(jc4_math_tests PRIVATE src "${REF}/src")
 target_compile_definitions(jc4_math_tests PRIVATE NOMINMAX)
 add_test(NAME jc4_math COMMAND jc4_math_tests)
endif()
