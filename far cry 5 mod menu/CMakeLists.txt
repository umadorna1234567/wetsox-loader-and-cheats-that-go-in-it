cmake_minimum_required(VERSION 3.20)
project(FC5Menu LANGUAGES C CXX)
set(FC5_DLL_NAME "FC5Menu" CACHE STRING "DLL output basename")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(fc5_core src/targeting.cpp)
target_include_directories(fc5_core PUBLIC include)
if(MSVC)
  target_compile_options(fc5_core PRIVATE /W4 /permissive-)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE fc5_core)
  add_test(NAME targeting COMMAND core_tests)
endif()
if(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Far Cry 5 requires the x64 build.")
  endif()
  add_library(fc5_menu STATIC src/menu_win32.cpp)
  target_compile_definitions(fc5_menu PUBLIC UNICODE _UNICODE NOMINMAX)
  target_link_libraries(fc5_menu PUBLIC fc5_core comdlg32)
  add_executable(FC5MenuPreview WIN32 src/preview_main.cpp)
  target_link_libraries(FC5MenuPreview PRIVATE fc5_menu)
  add_library(FC5Menu SHARED src/dll_main.cpp)
  set_target_properties(FC5Menu PROPERTIES OUTPUT_NAME "${FC5_DLL_NAME}")
  target_compile_definitions(FC5Menu PRIVATE FC5_DLL_BUILD)
  add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
  add_library(imgui STATIC
    third_party/imgui/imgui.cpp third_party/imgui/imgui_draw.cpp
    third_party/imgui/imgui_tables.cpp third_party/imgui/imgui_widgets.cpp
    third_party/imgui/backends/imgui_impl_win32.cpp
    third_party/imgui/backends/imgui_impl_dx11.cpp)
  target_include_directories(imgui PUBLIC third_party/imgui third_party/imgui/backends)
  target_compile_definitions(imgui PRIVATE UNICODE _UNICODE NOMINMAX)
  target_link_libraries(imgui PUBLIC d3d11 dxgi d3dcompiler dwmapi)
  add_library(fc5_input STATIC src/input.cpp)
  target_include_directories(fc5_input PUBLIC include)
  target_compile_definitions(fc5_input PUBLIC UNICODE _UNICODE NOMINMAX)
  target_link_libraries(fc5_input PUBLIC minhook dinput8 dxguid)
  target_sources(FC5Menu PRIVATE src/overlay.cpp src/runtime.cpp)
  target_link_libraries(FC5Menu PRIVATE fc5_menu imgui minhook bcrypt fc5_input)
  add_custom_command(TARGET FC5Menu POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:FC5Menu>/licenses"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui/LICENSE.txt" "$<TARGET_FILE_DIR:FC5Menu>/licenses/imgui.txt"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
      "${CMAKE_CURRENT_SOURCE_DIR}/third_party/minhook/LICENSE.txt" "$<TARGET_FILE_DIR:FC5Menu>/licenses/minhook.txt")
  add_executable(FC5MenuLoader src/loader.cpp)
  target_include_directories(FC5MenuLoader PRIVATE include)
  target_compile_definitions(FC5MenuLoader PRIVATE UNICODE _UNICODE NOMINMAX FC5_DLL_FILENAME=L"${FC5_DLL_NAME}.dll")
  if(BUILD_TESTING)
    add_test(NAME menu_smoke COMMAND FC5MenuPreview --smoke-test)
    add_executable(input_tests tests/input_tests.cpp)
    target_link_libraries(input_tests PRIVATE fc5_input)
    add_test(NAME input_capture COMMAND input_tests)
    add_executable(dll_tests tests/dll_tests.cpp)
    target_compile_definitions(dll_tests PRIVATE UNICODE _UNICODE NOMINMAX)
    add_dependencies(dll_tests FC5Menu)
    add_test(NAME dll_lifecycle COMMAND dll_tests $<TARGET_FILE:FC5Menu>)
    add_executable(FC5OverlayHost tests/overlay_host.cpp)
    target_include_directories(FC5OverlayHost PRIVATE include)
    target_compile_definitions(FC5OverlayHost PRIVATE UNICODE _UNICODE NOMINMAX)
    target_link_libraries(FC5OverlayHost PRIVATE d3d11 dxgi)
    add_dependencies(FC5OverlayHost FC5Menu)
    add_test(NAME overlay_render COMMAND FC5OverlayHost $<TARGET_FILE:FC5Menu>)
    set_tests_properties(overlay_render PROPERTIES TIMEOUT 40)
    find_package(Python3 COMPONENTS Interpreter REQUIRED)
    add_test(NAME loader_integration COMMAND ${Python3_EXECUTABLE}
      ${CMAKE_CURRENT_SOURCE_DIR}/tests/loader_integration.py
      $<TARGET_FILE:FC5OverlayHost> $<TARGET_FILE:FC5MenuLoader>)
    set_tests_properties(loader_integration PROPERTIES TIMEOUT 90)
  endif()
endif()
